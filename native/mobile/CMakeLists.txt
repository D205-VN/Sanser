cmake_minimum_required(VERSION 3.22)
project(sanser_mobile VERSION 2.3.7 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
include(../common/opus.cmake)
add_subdirectory(../protocol ${CMAKE_CURRENT_BINARY_DIR}/protocol)
set(MOBILE_SOURCES client.cpp audio_playout.cpp ../desktop/transport.cpp ../desktop/voice.cpp)
if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Mobile Direct currently builds Android ARM64 only")
  endif()
  add_library(sanser_mobile_p2p STATIC IMPORTED)
  set_target_properties(sanser_mobile_p2p PROPERTIES IMPORTED_LOCATION
    "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/mobile/build/rust-p2p/aarch64-linux-android/release/libsanser_mobile_p2p.a")
  add_library(sanser_mobile SHARED ${MOBILE_SOURCES} android_jni.cpp)
  target_link_libraries(sanser_mobile PRIVATE sanser_mobile_p2p log android dl m)
else()
  add_library(sanser_mobile STATIC ${MOBILE_SOURCES})
endif()
target_include_directories(sanser_mobile PUBLIC . ../desktop ../protocol ../common)
target_link_libraries(sanser_mobile PUBLIC sanser-snv2 sanser-audio-codec)
option(SANSER_MOBILE_TESTS "Build mobile C ABI interoperability tests" OFF)
if(SANSER_MOBILE_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(sanser-mobile-gamepad-test gamepad_capture_test.cpp)
  target_include_directories(sanser-mobile-gamepad-test PRIVATE .)
  add_test(NAME sanser-mobile-gamepad COMMAND sanser-mobile-gamepad-test)
  add_executable(sanser-mobile-media-thread-test ../common/media_thread_test.cpp)
  add_test(NAME sanser-mobile-media-thread COMMAND sanser-mobile-media-thread-test)
  add_executable(sanser-mobile-pacing-test ../desktop/mobile_pacing_test.cpp)
  target_link_libraries(sanser-mobile-pacing-test PRIVATE sanser_mobile)
  add_test(NAME sanser-mobile-pacing COMMAND sanser-mobile-pacing-test)
  add_executable(sanser-mobile-test client_test.cpp)
  target_link_libraries(sanser-mobile-test PRIVATE sanser_mobile)
  add_test(NAME sanser-mobile-contract COMMAND sanser-mobile-test)
  add_executable(sanser-mobile-pcm-test audio_playout_test.cpp)
  target_link_libraries(sanser-mobile-pcm-test PRIVATE sanser_mobile)
  add_test(NAME sanser-mobile-pcm COMMAND sanser-mobile-pcm-test)
  add_executable(sanser-mobile-rate-test ../desktop/rate_control_test.cpp)
  target_link_libraries(sanser-mobile-rate-test PRIVATE sanser_mobile)
  add_test(NAME sanser-mobile-rate COMMAND sanser-mobile-rate-test)
endif()
