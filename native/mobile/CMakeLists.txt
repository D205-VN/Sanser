cmake_minimum_required(VERSION 3.22)
project(sanser_mobile VERSION 2.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
include(../common/opus.cmake)
add_subdirectory(../protocol ${CMAKE_CURRENT_BINARY_DIR}/protocol)
set(MOBILE_SOURCES client.cpp ../desktop/transport.cpp ../desktop/voice.cpp)
if(ANDROID)
  add_library(sanser_mobile SHARED ${MOBILE_SOURCES} android_jni.cpp)
  target_link_libraries(sanser_mobile PRIVATE log android)
else()
  add_library(sanser_mobile STATIC ${MOBILE_SOURCES})
endif()
target_include_directories(sanser_mobile PUBLIC . ../desktop ../protocol ../common)
target_link_libraries(sanser_mobile PUBLIC sanser-snv2 sanser-audio-codec)
option(SANSER_MOBILE_TESTS "Build mobile C ABI interoperability tests" OFF)
if(SANSER_MOBILE_TESTS AND NOT ANDROID)
  enable_testing()
  add_executable(sanser-mobile-test client_test.cpp)
  target_link_libraries(sanser-mobile-test PRIVATE sanser_mobile)
  add_test(NAME sanser-mobile-contract COMMAND sanser-mobile-test)
endif()
