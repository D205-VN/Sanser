cmake_minimum_required(VERSION 3.20)
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum supported macOS version")
endif()
project(sanser_desktop_native VERSION 2.2.5 LANGUAGES CXX)
set(SANSER_APP_VERSION "${PROJECT_VERSION}" CACHE STRING "Sanser version")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET sanser-snv2)
  add_subdirectory(../protocol ${CMAKE_CURRENT_BINARY_DIR}/protocol)
endif()
add_library(sanser-desktop-transport STATIC transport.cpp broadcast.cpp)
target_link_libraries(sanser-desktop-transport PUBLIC sanser-snv2)
target_include_directories(sanser-desktop-transport PUBLIC . ../common)
# Transport is shared; each OS owns its capture, codec, window and input backend.
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  add_subdirectory(../platforms/macos ${CMAKE_CURRENT_BINARY_DIR}/platform-macos)
  add_executable(sanser-host-macos main.cpp)
  target_compile_definitions(sanser-host-macos PRIVATE SANSER_ENGINE_HOST=1 SANSER_VERSION="${SANSER_APP_VERSION}")
  target_link_libraries(sanser-host-macos PRIVATE sanser-desktop-native)
elseif(WIN32)
  add_subdirectory(../platforms/windows ${CMAKE_CURRENT_BINARY_DIR}/platform-windows)
  target_link_libraries(sanser-desktop-transport PUBLIC ws2_32 bcrypt winmm)
  add_executable(sanser-client-windows main.cpp)
  target_compile_definitions(sanser-client-windows PRIVATE SANSER_VERSION="${SANSER_APP_VERSION}")
  target_link_libraries(sanser-client-windows PRIVATE sanser-desktop-native)
endif()
include(CTest)

  add_executable(sanser-decode-worker-test decode_worker_test.cpp)
  target_link_libraries(sanser-decode-worker-test PRIVATE sanser-desktop-transport)
  add_test(NAME sanser-decode-worker-test COMMAND sanser-decode-worker-test)
  set_tests_properties(sanser-decode-worker-test PROPERTIES TIMEOUT 10)
  add_executable(sanser-broadcast-policy-test broadcast_policy_test.cpp)
  add_test(NAME sanser-broadcast-policy-test COMMAND sanser-broadcast-policy-test)
  add_executable(sanser-broadcast-test broadcast_test.cpp)
  target_link_libraries(sanser-broadcast-test PRIVATE sanser-desktop-transport)
  add_test(NAME sanser-broadcast-test COMMAND sanser-broadcast-test)
  set_tests_properties(sanser-broadcast-test PROPERTIES TIMEOUT 45)
  add_executable(sanser-desktop-transport-bench transport_bench.cpp)
  target_link_libraries(sanser-desktop-transport-bench PRIVATE sanser-desktop-transport)
  add_executable(sanser-desktop-transport-test transport_test.cpp)
  target_link_libraries(sanser-desktop-transport-test PRIVATE sanser-desktop-transport)
  add_test(NAME sanser-desktop-transport-test COMMAND sanser-desktop-transport-test)
  add_executable(sanser-desktop-loopback-test loopback_test.cpp)
  target_link_libraries(sanser-desktop-loopback-test PRIVATE sanser-desktop-transport)
  add_test(NAME sanser-desktop-loopback-test COMMAND sanser-desktop-loopback-test)
  set_tests_properties(sanser-desktop-loopback-test PROPERTIES TIMEOUT 20)
  add_executable(sanser-input-recovery-test input_recovery_test.cpp)
  target_link_libraries(sanser-input-recovery-test PRIVATE sanser-desktop-transport)
  add_test(NAME sanser-input-recovery-test COMMAND sanser-input-recovery-test)
  set_tests_properties(sanser-input-recovery-test PROPERTIES TIMEOUT 15)
  if(WIN32)
    add_executable(sanser-windows-presenter-test ../platforms/windows/window_presenter_test.cpp)
    target_include_directories(sanser-windows-presenter-test PRIVATE ../platforms/windows)
    target_link_libraries(sanser-windows-presenter-test PRIVATE sanser-desktop-native)
    add_test(NAME sanser-windows-presenter-test COMMAND sanser-windows-presenter-test)
    set_tests_properties(sanser-windows-presenter-test PROPERTIES TIMEOUT 10)
  endif()
  if(APPLE)
    add_executable(sanser-macos-capture-startup-test ../platforms/macos/capture_startup_test.mm)
    target_compile_options(sanser-macos-capture-startup-test PRIVATE -fobjc-arc -Wall -Wextra)
    target_include_directories(sanser-macos-capture-startup-test PRIVATE ../platforms/macos)
    target_link_libraries(sanser-macos-capture-startup-test PRIVATE sanser-desktop-native)
    add_test(NAME sanser-macos-capture-startup-test COMMAND sanser-macos-capture-startup-test)
    set_tests_properties(sanser-macos-capture-startup-test PROPERTIES TIMEOUT 10)
    target_compile_definitions(sanser-native-macos PRIVATE SANSER_CODEC_TEST=1)
    add_executable(sanser-macos-codec-test macos_codec_test.cpp)
    target_link_libraries(sanser-macos-codec-test PRIVATE sanser-desktop-native)
    add_test(NAME sanser-macos-codec-test COMMAND sanser-macos-codec-test)
    set_tests_properties(sanser-macos-codec-test PROPERTIES TIMEOUT 20)
  endif()
endif()
