cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(sanser_windows_text_input LANGUAGES CXX)
  if(NOT WIN32)
    message(FATAL_ERROR "Use a Windows toolchain for the native EDIT tests")
  endif()
  include(CTest)
endif()

add_library(sanser-windows-text-input STATIC ../ime_panel.cpp ../shortcuts_panel.cpp ../controller_input.cpp)
target_compile_features(sanser-windows-text-input PUBLIC cxx_std_20)
target_include_directories(sanser-windows-text-input PUBLIC .. PRIVATE ../../../common)
target_link_libraries(sanser-windows-text-input PUBLIC user32 gdi32 comctl32 imm32)
target_compile_definitions(sanser-windows-text-input PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(sanser-windows-text-input PRIVATE /utf-8 /W4)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND BUILD_TESTING)
  add_executable(sanser-windows-controller-test ../controller_input_test.cpp)
  target_link_libraries(sanser-windows-controller-test PRIVATE sanser-windows-text-input)
  add_test(NAME sanser-windows-controller-test COMMAND sanser-windows-controller-test)
  add_executable(sanser-controller-session-test ../../../desktop/controller_session_test.cpp)
  target_compile_features(sanser-controller-session-test PRIVATE cxx_std_20)
  add_test(NAME sanser-controller-session-test COMMAND sanser-controller-session-test)
  add_executable(sanser-gamepad-test ../../../desktop/gamepad_test.cpp ../gamepad.cpp)
  target_compile_features(sanser-gamepad-test PRIVATE cxx_std_20)
  target_compile_definitions(sanser-gamepad-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_include_directories(sanser-gamepad-test PRIVATE ../../../desktop)
  add_test(NAME sanser-gamepad-test COMMAND sanser-gamepad-test)
  set_tests_properties(sanser-gamepad-test PROPERTIES TIMEOUT 10)
  add_executable(sanser-windows-shortcuts-test ../shortcuts_panel_test.cpp)
  target_link_libraries(sanser-windows-shortcuts-test PRIVATE sanser-windows-text-input)
  add_test(NAME sanser-windows-shortcuts-test COMMAND sanser-windows-shortcuts-test)
  add_executable(sanser-windows-ime-test ../ime_panel_test.cpp)

  add_test(NAME sanser-ime-commit-test COMMAND sanser-ime-commit-test)
  if(MSVC)
    target_compile_options(sanser-windows-ime-test PRIVATE /utf-8 /W4)
    target_compile_options(sanser-ime-commit-test PRIVATE /utf-8 /W4)
  endif()
endif()
